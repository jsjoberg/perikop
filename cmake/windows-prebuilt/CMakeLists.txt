cmake_minimum_required(VERSION 3.24)
# This project never enables a compiler or FetchContent.
file(READ "${CMAKE_BINARY_DIR}/windows-handoff.json" perikop_handoff)
string(JSON perikop_schema GET "${perikop_handoff}" schema)
if(NOT perikop_schema EQUAL 1)
    message(FATAL_ERROR "Unsupported Windows handoff schema: ${perikop_schema}")
endif()
string(JSON perikop_project_version GET "${perikop_handoff}" project_version)
string(JSON perikop_prebuilt_version GET "${perikop_handoff}" version)
project(perikop VERSION "${perikop_project_version}" LANGUAGES NONE)
get_filename_component(perikop_source_dir "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

function(perikop_local_value output value)
    string(REPLACE "@BUILD@" "${CMAKE_BINARY_DIR}" value "${value}")
    string(REPLACE "@SOURCE@" "${perikop_source_dir}" value "${value}")
    string(REPLACE "@CMAKE@" "${CMAKE_COMMAND}" value "${value}")
    set(${output} "${value}" PARENT_SCOPE)
endfunction()

include(CTest)
string(JSON perikop_test_count LENGTH "${perikop_handoff}" tests)
if(perikop_test_count LESS 1)
    message(FATAL_ERROR "Windows handoff contains no tests")
endif()
math(EXPR perikop_last_test "${perikop_test_count} - 1")
foreach(index RANGE ${perikop_last_test})
    string(JSON test GET "${perikop_handoff}" tests ${index})
    string(JSON name GET "${test}" name)
    string(JSON argument_count LENGTH "${test}" command)
    if(argument_count LESS 1)
        message(FATAL_ERROR "Prebuilt test has no command: ${name}")
    endif()
    math(EXPR last_argument "${argument_count} - 1")
    set(command)
    foreach(argument_index RANGE ${last_argument})
        string(JSON argument GET "${test}" command ${argument_index})
        perikop_local_value(argument "${argument}")
        # Keep semicolons inside arguments, not as list separators.
        string(REPLACE ";" "\\;" argument "${argument}")
        list(APPEND command "${argument}")
    endforeach()
    list(GET command 0 executable)
    if(NOT EXISTS "${executable}")
        message(FATAL_ERROR "Prebuilt test executable is missing: ${executable}")
    endif()
    add_test(NAME "${name}" COMMAND ${command})
    string(JSON property_count LENGTH "${test}" properties)
    if(property_count GREATER 0)
        math(EXPR last_property "${property_count} - 1")
        foreach(property_index RANGE ${last_property})
            string(JSON property GET "${test}" properties ${property_index})
            string(JSON property_name GET "${property}" name)
            string(JSON property_type TYPE "${property}" value)
            if(property_type STREQUAL "ARRAY")
                string(JSON value_count LENGTH "${property}" value)
                set(value)
                if(value_count GREATER 0)
                    math(EXPR last_value "${value_count} - 1")
                    foreach(value_index RANGE ${last_value})
                        string(JSON item GET "${property}" value ${value_index})
                        perikop_local_value(item "${item}")
                        string(REPLACE ";" "\\;" item "${item}")
                        list(APPEND value "${item}")
                    endforeach()
                endif()
            else()
                string(JSON value GET "${property}" value)
                perikop_local_value(value "${value}")
            endif()
            set_tests_properties("${name}" PROPERTIES "${property_name}" "${value}")
        endforeach()
    endif()
endforeach()

foreach(file bin/perikop.exe bin/velopack_libc.dll bin/resources/LICENSE)
    if(NOT EXISTS "${CMAKE_BINARY_DIR}/${file}")
        message(FATAL_ERROR "Prebuilt application file is missing: ${file}")
    endif()
endforeach()
# Linux strips perikop.exe before transfer. No Windows strip tool is needed.
install(PROGRAMS "${CMAKE_BINARY_DIR}/bin/perikop.exe" DESTINATION bin)
install(FILES "${CMAKE_BINARY_DIR}/bin/velopack_libc.dll" DESTINATION bin)
install(DIRECTORY "${CMAKE_BINARY_DIR}/bin/resources/" DESTINATION bin/resources)
include("${perikop_source_dir}/cmake/package.cmake")
