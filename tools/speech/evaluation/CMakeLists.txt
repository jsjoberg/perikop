cmake_minimum_required(VERSION 3.20)
project(bjorn_unit_selection LANGUAGES CXX)
add_executable(bjorn-unit-render unit_render.cpp)
add_executable(bjorn-grain-render grain_render.cpp)
foreach(target bjorn-unit-render bjorn-grain-render)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
