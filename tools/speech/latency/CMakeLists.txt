cmake_minimum_required(VERSION 3.24)
project(perikop_speech_latency LANGUAGES C CXX)
# Reuse the application's existing dependencies. No downloads or new speech engine.
set(ORT_ROOT "" CACHE PATH "Existing ONNX Runtime installation with include/ and lib/")
set(SONIC_ROOT "" CACHE PATH "Existing Sonic source directory containing sonic.c")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ORT_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(ORT_LIBRARY NAMES onnxruntime PATHS "${ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
if(NOT EXISTS "${SONIC_ROOT}/sonic.c")
    message(FATAL_ERROR "Set SONIC_ROOT to the existing Sonic source directory")
endif()
find_package(SQLite3 3.37 REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
# Instrument build-directory copies, preserving application sources and PCM behavior.
file(READ "${ROOT}/src/speech/portable_speech.cpp" worker)
foreach(anchor IN ITEMS "kokoro = std::make_unique<Kokoro>(pack_);" "samples = voice ==" "store.save(revision, language, text, *samples);")
    string(FIND "${worker}" "${anchor}" found)
    if(found EQUAL -1)
        message(FATAL_ERROR "Worker changed; review latency instrumentation: ${anchor}")
    endif()
endforeach()
string(REPLACE "kokoro = std::make_unique<Kokoro>(pack_);" "speech_probe_trace(\"model_load_begin\"); kokoro = std::make_unique<Kokoro>(pack_); speech_probe_trace(\"model_load_end\");" worker "${worker}")
string(REPLACE "samples = voice ==" "speech_probe_trace(\"generate_begin\"); samples = voice ==" worker "${worker}")
string(REPLACE "model_seconds +=" "speech_probe_trace(\"generate_end\"); model_seconds +=" worker "${worker}")
string(REPLACE "store.save(revision, language, text, *samples);" "speech_probe_trace(\"cache_save_begin\"); store.save(revision, language, text, *samples); speech_probe_trace(\"cache_save_end\");" worker "${worker}")
string(REPLACE "std::vector<Part> parts;" "speech_probe_trace(\"parts_begin\"); std::vector<Part> parts;" worker "${worker}")
string(REPLACE "double remaining_weight =" "speech_probe_trace(\"parts_end\"); double remaining_weight =" worker "${worker}")
string(REPLACE "ready_ = std::min(1.0, buffered / target);" "speech_probe_trace(\"buffer_gate\", target, buffered); ready_ = std::min(1.0, buffered / target);" worker "${worker}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/worker.cpp" "#include \"probe_hooks.hpp\"\n${worker}")
file(READ "${ROOT}/src/speech/kokoro_audio.cpp" audio)
string(FIND "${audio}" "options.SetIntraOpNumThreads(4);" found)
if(found EQUAL -1)
    message(FATAL_ERROR "Audio thread configuration changed; review instrumentation")
endif()
string(REPLACE "options.SetIntraOpNumThreads(4);" "options.SetIntraOpNumThreads(speech_probe_threads());" audio "${audio}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/audio.cpp" "#include \"probe_hooks.hpp\"\n${audio}")
add_executable(speech-latency probe.cpp "${CMAKE_CURRENT_BINARY_DIR}/worker.cpp" "${CMAKE_CURRENT_BINARY_DIR}/audio.cpp"
    "${ROOT}/src/speech/kokoro_text.cpp" "${ROOT}/src/speech/compact_voice.cpp"
    "${ROOT}/src/speech/swedish_numbers.cpp" "${ROOT}/src/speech/speech.cpp"
    "${ROOT}/src/speech/tempo.cpp" "${ROOT}/src/storage/sqlite.cpp"
    "${ROOT}/src/storage/speech_cache.cpp" "${SONIC_ROOT}/sonic.c")
target_compile_features(speech-latency PRIVATE cxx_std_23)
target_include_directories(speech-latency PRIVATE "${ROOT}/src" "${CMAKE_CURRENT_SOURCE_DIR}" "${ORT_INCLUDE}" "${SONIC_ROOT}")
target_link_libraries(speech-latency PRIVATE "${ORT_LIBRARY}" SQLite::SQLite3 Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(speech-latency PROPERTIES BUILD_RPATH "${ORT_ROOT}/lib")
