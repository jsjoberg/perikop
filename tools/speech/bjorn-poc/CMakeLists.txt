cmake_minimum_required(VERSION 3.24)
project(bjorn_speech LANGUAGES C CXX)
# Independent experiment: deliberately does not configure or link the application.
add_executable(bjorn-render render.c)
set_target_properties(bjorn-render PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(NOT MSVC)
    target_compile_options(bjorn-render PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(bjorn-render PRIVATE m)
endif()
get_filename_component(bjorn_source_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(bjorn-compact STATIC "${bjorn_source_root}/src/speech/compact_voice.cpp"
    "${bjorn_source_root}/src/speech/swedish_numbers.cpp")
target_include_directories(bjorn-compact PUBLIC "${bjorn_source_root}/src")
target_compile_features(bjorn-compact PUBLIC cxx_std_23)
add_executable(bjorn-say ../compact_voice_probe.cpp)
target_link_libraries(bjorn-say PRIVATE bjorn-compact)
include(CTest)
if(BUILD_TESTING)
    add_executable(bjorn-compact-tests "${bjorn_source_root}/tests/compact_voice_test.cpp")
    target_link_libraries(bjorn-compact-tests PRIVATE bjorn-compact)
    add_test(NAME compact-voice COMMAND bjorn-compact-tests
        "${bjorn_source_root}/resources/voices/bjorn-lite-v1" "${CMAKE_CURRENT_BINARY_DIR}/test-fixture")
endif()
